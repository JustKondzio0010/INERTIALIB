local function CreateTween(instance, properties, duration, easingStyle, easingDirection)
    if not instance then return end
    local tween = ts:Create(
        instance,
        TweenInfo.new(
            duration or animationspeed.Normal,
            easingStyle or Enum.EasingStyle.Quart,
            easingDirection or Enum.EasingDirection.Out
        ),
        properties
    )
    tween:Play()
    return tween
end

local function CreateInstance(className, properties)
    local instance = Instance.new(className)
    for property, value in pairs(properties) do
        if property ~= "Parent" then
            instance[property] = value
        end
    end
    if properties.Parent then
        instance.Parent = properties.Parent
    end
    return instance
end

local function CreateCorner(parent, radius)
    return CreateInstance("UICorner", {
        CornerRadius = UDim.new(0, radius or 6),
        Parent = parent
    })
end

local function IsMobileDevice()
    if not ui.TouchEnabled then
        return false
    end
    if ui.MouseEnabled == false then
        return true
    end
    if workspace.CurrentCamera then
        local v = workspace.CurrentCamera.ViewportSize
        if v and v.X > 0 and v.X < 900 then
            return true
        end
    end
    return false
end

local function EnsureTooltip(screenGui)
    if TooltipFrame then return end
    TooltipFrame = CreateInstance("Frame", {
        Name = "InertiaTooltip",
        BackgroundColor3 = c.Secondary,
        BackgroundTransparency = 0.05,
        BorderSizePixel = 0,

        BackgroundTransparency = 1,

        Parent = TooltipFrame

local function AttachTooltip(interactiveElement, tooltipText, screenGui)
    if not tooltipText or tooltipText == "" then return end
    EnsureTooltip(screenGui)
    local hoverConn1, hoverConn2, hoverConn3
    local hoverActive = false

    hoverConn1 = interactiveElement.MouseEnter:Connect(function()
        hoverActive = true
        task.delay(0.35, function()
            if hoverActive and TooltipFrame and TooltipLabel then
                TooltipLabel.Text = tooltipText
                local mousePos = ui:GetMouseLocation()
                TooltipFrame.Position = UDim2.new(0, mousePos.X + 12, 0, mousePos.Y + 12)
                TooltipFrame.Visible = true
                CreateTween(TooltipFrame, {BackgroundTransparency = 0.05}, animationspeed.Fast)
            end
        end)
    end)

    hoverConn2 = interactiveElement.MouseLeave:Connect(function()
        hoverActive = false
        if TooltipFrame then
            TooltipFrame.Visible = false
        end
    end)

    hoverConn3 = interactiveElement.InputChanged:Connect(function(input)
        if hoverActive and TooltipFrame and TooltipFrame.Visible and input.UserInputType == Enum.UserInputType.MouseMovement then
            local mousePos = ui:GetMouseLocation()
            TooltipFrame.Position = UDim2.new(0, mousePos.X + 12, 0, mousePos.Y + 12)
        end
    end)

    table.insert(Connections, hoverConn1)
    table.insert(Connections, hoverConn2)
    table.insert(Connections, hoverConn3)
end

-- ============================================================================
-- DRAGGABLE IMPLEMENTATION
-- ============================================================================
local function MakeDraggable(frame, handle)
    local dragging = false
    local dragInput, dragStart, startPos
    handle = handle or frame

    local function OnInputBegan(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            if input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
            local connection
            connection = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if connection then
                        connection:Disconnect()
                    end
                end
            end)
        end
    end

    local function OnInputChanged(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            dragInput = input
        end
    end

    table.insert(Connections, handle.InputBegan:Connect(OnInputBegan))
    table.insert(Connections, handle.InputChanged:Connect(OnInputChanged))

    table.insert(Connections, ui.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            local newX = startPos.X.Offset + delta.X
            local newY = startPos.Y.Offset + delta.Y

            if workspace.CurrentCamera then
                local viewport = workspace.CurrentCamera.ViewportSize
                local inset = gs:GetGuiInset()
                local size = frame.AbsoluteSize
                local anchor = frame.AnchorPoint
                local pad = 8
                local scalePxX = startPos.X.Scale * viewport.X
                local scalePxY = startPos.Y.Scale * viewport.Y

                local minTLX = inset.X + pad
                local maxTLX = viewport.X - size.X - pad
                local minTLY = inset.Y + pad
                local maxTLY = viewport.Y - size.Y - pad

                local minOffX = minTLX - scalePxX + (anchor.X * size.X)
                local maxOffX = maxTLX - scalePxX + (anchor.X * size.X)
                local minOffY = minTLY - scalePxY + (anchor.Y * size.Y)
                local maxOffY = maxTLY - scalePxY + (anchor.Y * size.Y)

                newX = math.clamp(newX, minOffX, maxOffX)
                newY = math.clamp(newY, minOffY, maxOffY)
            end

            frame.Position = UDim2.new(startPos.X.Scale, newX, startPos.Y.Scale, newY)
        end
    end))
end

local function DisconnectAll()
    for _, connection in pairs(Connections) do
        if typeof(connection) == "RBXScriptConnection" then
            connection:Disconnect()
        end
    end
    Connections = {}
end

local function GetConfigFolder(configName)
    return "InertiaConfigs/" .. configName
end

local function EnsureConfigFolder()
    if isfolder and not isfolder("InertiaConfigs") then
        makefolder("InertiaConfigs")
    end
    if isfolder and not isfolder("AcrylicConfigs") then
        makefolder("AcrylicConfigs")
    end
end

local function GetAvailableConfigs()
    local configs = {}
    if isfolder and listfiles then
        EnsureConfigFolder()
        local folders = {"InertiaConfigs", "AcrylicConfigs"}
        for _, folder in ipairs(folders) do
            if isfolder(folder) then
                local files = listfiles(folder)
                for _, file in ipairs(files) do
                    local name = file:match(folder .. "/(.+)%.json$") or file:match(folder .. "\\(.+)%.json$")
                    if name and not table.find(configs, name) then
                        table.insert(configs, name)
                    end
                end
            end
        end
    end
    return configs
end

local AcrylicBlur = {}

function AcrylicBlur.new(object)
    local self = setmetatable({
        _object = object,
        _folder = nil,
        _root = nil,
        _frame = nil,
        _dof = nil,
        _enabled = true
    }, AcrylicBlur)

    return self
end

function AcrylicBlur:_CreateDepthOfField()
    local existingDOF = lg:FindFirstChild("InertiaBlur") or lg:FindFirstChild("AcrylicBlur")
    if existingDOF then
        existingDOF:Destroy()
    end
    local existingBlur = lg:FindFirstChild("InertiaBlurEffect") or lg:FindFirstChild("AcrylicBlurEffect")
    if existingBlur then
        existingBlur:Destroy()
    end
    
    local dof = CreateInstance("DepthOfFieldEffect", {
        Name = "InertiaBlur",
        FarIntensity = 0,
        FocusDistance = 0.05,
        InFocusRadius = 0.1,
        NearIntensity = 0.5,
        Parent = lg
    })
    
    self._dof = dof
    return dof
end

function AcrylicBlur:_CreateFolder()
    local existingFolder = workspace.CurrentCamera:FindFirstChild("InertiaBlur") or workspace.CurrentCamera:FindFirstChild("AcrylicBlur")
    if existingFolder then
        existingFolder:Destroy()
    end
    self._folder = CreateInstance("Folder", {
        Name = "InertiaBlur",
        Parent = workspace.CurrentCamera
    })
end

function AcrylicBlur:_CreateRoot()
    local part = CreateInstance("Part", {
        Name = "Root",
        Color = Color3.new(0, 0, 0),
        Material = Enum.Material.Glass,
        Size = Vector3.new(1, 1, 0),
        Anchored = true,
        CanCollide = false,
        CanQuery = false,
        Locked = true,
        CastShadow = false,
        Transparency = 0.95,
        Parent = self._folder
    })
    CreateInstance("SpecialMesh", {
        MeshType = Enum.MeshType.Brick,
        Parent = part
    })
    self._root = part
end

function AcrylicBlur:_CreateFrame()
    self._frame = CreateInstance("Frame", {
        Size = UDim2.new(1, 0, 1, 0),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundTransparency = 1,
        Parent = self._object
    })
end

function AcrylicBlur:_Render(distance)
    distance = distance or 0.001
    local positions = {
        top_left = Vector2.new(),
        top_right = Vector2.new(),
        bottom_right = Vector2.new()
    }

    local function ViewportToWorld(location, dist)
        local ray = workspace.CurrentCamera:ScreenPointToRay(location.X, location.Y)
        return ray.Origin + ray.Direction * dist
    end

    local function GetOffset()
        local viewY = workspace.CurrentCamera.ViewportSize.Y
        return (viewY / 2560) * 24 + 4
    end

    local function CalculatePositions()
        local offset = GetOffset()
        local framePos = self._frame.AbsolutePosition
        local frameSize = self._frame.AbsoluteSize

        positions.top_left = ViewportToWorld(framePos + Vector2.new(0, offset), distance)
        positions.top_right = ViewportToWorld(framePos + Vector2.new(frameSize.X, offset), distance)
        positions.bottom_right = ViewportToWorld(framePos + Vector2.new(frameSize.X, frameSize.Y + offset), distance)
    end

    local function UpdateMesh()
        if not self._root then return end
        local width = (positions.top_right - positions.top_left).Magnitude
        local height = (positions.top_right - positions.bottom_right).Magnitude
        self._root.CFrame = CFrame.fromMatrix(
            (positions.top_left + positions.bottom_right) / 2,
            (positions.top_right - positions.top_left).Unit,
            (positions.top_right - positions.bottom_right).Unit
        )
        local mesh = self._root:FindFirstChildOfClass("SpecialMesh")
        if mesh then
            mesh.Scale = Vector3.new(width, height, 0)
        end
    end

    local renderConnection
    renderConnection = rs.RenderStepped:Connect(function()
        if not self._enabled or not self._root or not self._frame then
            if renderConnection then
                renderConnection:Disconnect()
            end
            return
        end
        CalculatePositions()
        UpdateMesh()
    end)
    table.insert(Connections, renderConnection)
end

function AcrylicBlur:_Initialize()
    self:_CreateFolder()
    self:_CreateRoot()
    self:_CreateFrame()
    self:_CreateDepthOfField()
    self:_Render()
end

function AcrylicBlur:SetEnabled(enabled)
    self._enabled = enabled
    if self._root then
        self._root.Transparency = enabled and 0.95 or 1
    end
    if self._dof then
        self._dof.Enabled = enabled
    end
end

function AcrylicBlur:Destroy()
    if self._folder then
        self._folder:Destroy()
    end
    local dof = lg:FindFirstChild("InertiaBlur") or lg:FindFirstChild("AcrylicBlur")
    if dof then
        dof:Destroy()
    end
    local blur = lg:FindFirstChild("InertiaBlurEffect") or lg:FindFirstChild("AcrylicBlurEffect")
    if blur then
        blur:Destroy()
    end
end

function Library:SetKeyTimeLeft(seconds)
    if not seconds then
        self._keyExpiresAt = nil
    else
        self._keyExpiresAt = os.time() + math.max(0, math.floor(tonumber(seconds) or 0))
    end
    self:_UpdateKeyTime()
end

function Library:SetKeyExpiresAt(unixTime)
    self._keyExpiresAt = tonumber(unixTime)
    self:_UpdateKeyTime()
end

function Library:ClearKey()
    self._keyExpiresAt = nil
    self:_UpdateKeyTime()
end

function Library:_FormatTimeLeft(seconds)
    seconds = math.max(0, math.floor(tonumber(seconds) or 0))
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    if h <= 0 then
        return string.format("%dm left", m)
    end
    return string.format("%dh %dm left", h, m)
end

function Library:_UpdateKeyTime()
    if not self._keyTimeHolder or not self._keyTimeLabel then return end
    if not self._keyExpiresAt then
        self._keyTimeHolder.Visible = false
        return
    end
    local left = self._keyExpiresAt - os.time()
    if left <= 0 then
        self._keyTimeLabel.Text = "Key expired"
        self._keyTimeHolder.Visible = true
        return
    end
    self._keyTimeLabel.Text = self:_FormatTimeLeft(left)
    self._keyTimeHolder.Visible = true
end

function Library:_StartKeyTimeLoop()
    task.spawn(function()
        while task.wait(1) do
            if not self or not self.container then break end
            self:_UpdateKeyTime()
        end
    end)
end

function Library:_LoadProfileBestKey()
    if not readfile or not isfile then return nil end
    local path = self:_KeyProfilePath()
    if not isfile(path) then return nil end
    local ok, content = pcall(readfile, path)
    if not ok or not content or content == "" then return nil end
    local dataOk, data = pcall(function() return hs:JSONDecode(content) end)
    if not dataOk or type(data) ~= "table" then return nil end
    local entries = {}
    if data.keys and type(data.keys) == "table" then
        for _, entry in ipairs(data.keys) do
            if entry and entry.key and entry.key ~= "" then
                table.insert(entries, entry)
            end
        end
    end
    if #entries == 0 then return nil end
    table.sort(entries, function(a, b)
        local tA = tonumber(a.saved_at) or 0
        local tB = tonumber(b.saved_at) or 0
        return tA > tB
    end)
    return entries[1].key
end

function Library:_VerifyKey(key)
    local req = self:_GetRequest()
    if not req then return false, "No HTTP client" end
    local hwid = self:_GenerateHWID()
    local fingerprint = self:_GenerateFingerprint()
    local payload = hs:JSONEncode({
        key = key,
        hwid = hwid,
        fingerprint = fingerprint,
        game_id = tostring(game.PlaceId),
        executor = identifyexecutor and identifyexecutor() or "Unknown"
    })
    local res = req({
        Url = self._apiUrl .. "/api/verify",
        Method = "POST",
        Headers = {["Content-Type"] = "application/json"},
        Body = payload
    })
    if not res or not res.Body then return false, "No response" end
    local ok, data = pcall(function() return hs:JSONDecode(res.Body) end)
    if not ok or type(data) ~= "table" then return false, "Bad JSON" end
    if data.valid == true or data.success == true then
        local expiresAt = data.expires_at or data.expiresAt
        local timeLeft = data.time_left or data.timeLeft
        if expiresAt then
            self:SetKeyExpiresAt(expiresAt)
        elseif timeLeft then
            self:SetKeyTimeLeft(timeLeft)
        end
        return true, data
    end
    return false, data.message or "Invalid"
end

function Library:_StartAutoKey()
    task.spawn(function()
        task.wait(1)
        local key = self:_LoadProfileBestKey()
        if not key then return end
        self:_VerifyKey(key)
    end)
end

function Library:_UpdateResponsiveScale()
    if not self.container then return end
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    if not viewport or viewport.X <= 0 or viewport.Y <= 0 then return end
    
    local w = s.v0rtexd.Width
    local h = s.v0rtexd.Height
    local margin = 32
    local availableW = math.max(100, viewport.X - margin)
    local availableH = math.max(100, viewport.Y - margin)
    local scaleX = availableW / w
    local scaleY = availableH / h
    local scale = math.min(1, scaleX, scaleY)
    
    if IsMobileDevice() then
        scale = math.min(scale, 0.9)
    end
    
    if self._uiScaleObject then
        self._uiScaleObject.Scale = scale * (self._manualScale or 1)
    end
end

function Library:_FilterTabsAndElements(query)
    for _, section in ipairs(self.sections) do
        local anyVisibleTab = false
        for _, tab in ipairs(section.tabs) do
            local tabMatches = (query == "") or (string.find(string.lower(tab.name), query, 1, true) ~= nil)
            tab.button.Visible = tabMatches
            if tabMatches then
                anyVisibleTab = true
            end
            -- Also search element titles inside the tab
            if tab.elements then
                for _, element in ipairs(tab.elements) do
                    if element.frame and element.name then
                        local elemMatches = (query == "") or (string.find(string.lower(element.name), query, 1, true) ~= nil)
                        element.frame.Visible = elemMatches
                    end
                end
            end
        end
        if section.frame then
            section.frame.Visible = (query == "") or anyVisibleTab
        end
    end
end

-- ============================================================================
-- RESIZE & MINIMIZE IMPLEMENTATION
-- ============================================================================
function Library:_SetupSmartResize(handle)
    local resizing = false
    local resizeStart, startSize, resizeInput

    table.insert(Connections, handle.MouseEnter:Connect(function()
        CreateTween(handle, {ImageColor3 = c.Text}, animationspeed.Fast)
    end))

    table.insert(Connections, handle.MouseLeave:Connect(function()
        if not resizing then
            CreateTween(handle, {ImageColor3 = c.TextFade}, animationspeed.Fast)
        end
    end))

    table.insert(Connections, handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            resizing = true
            resizeStart = input.Position
            startSize = Vector2.new(self.container.Size.X.Offset, self.container.Size.Y.Offset)
            self._originalHeight = startSize.Y
            resizeInput = input.UserInputType == Enum.UserInputType.Touch and input or nil

            local connection
            connection = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    resizing = false
                    CreateTween(handle, {ImageColor3 = c.TextFade}, animationspeed.Fast)
                    if connection then
                        connection:Disconnect()
                    end
                    resizeInput = nil
                end
            end)
        end
    end))

    table.insert(Connections, handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement and resizing and not resizeInput then
            resizeInput = input
        end
    end))

    table.insert(Connections, ui.InputChanged:Connect(function(input)
        if resizing and resizeInput and input == resizeInput and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - resizeStart
            local scale = 1
            if self._uiScaleObject then
                scale = scale * self._uiScaleObject.Scale
            end
            if self._animScaleObject then
                scale = scale * self._animScaleObject.Scale
            end
            scale = math.max(0.01, scale)

            local newWidth = math.clamp(startSize.X + (delta.X / scale), self._minSize.X, self._maxSize.X)
            local newHeight = math.clamp(startSize.Y + (delta.Y / scale), self._minSize.Y, self._maxSize.Y)
            self.container.Size = UDim2.new(0, newWidth, 0, newHeight)
            self._originalHeight = newHeight
        end
    end))
end

function Library:_ToggleMinimize()
    self.minimized = not self.minimized

        if self._acrylicBlur then

-- CONFIG REGISTER & IO

function Library:_RegisterConfigElement(id, elementType, getValue, setValue)
    self._configElements[id] = {
        type = elementType,
        get = getValue,
        set = setValue
    }
end

function Library:SaveConfig(configName)
    if not writefile then return end
    EnsureConfigFolder()
    local data = {}
    for id, elem in pairs(self._configElements) do
        local val = elem.get()
        if typeof(val) == "Color3" then
            data[id] = {val.R, val.G, val.B}
        elseif typeof(val) == "EnumItem" then
            data[id] = val.Name
        else
            data[id] = val
        end
    end
    local ok, encoded = pcall(function() return hs:JSONEncode(data) end)
    if ok and encoded then
        pcall(writefile, GetConfigFolder(configName) .. ".json", encoded)
        self:Notify({
            Title = "Config Saved",
            Description = "Saved configuration '" .. configName .. "'",
            Type = "Success",
            Duration = 2.5
        })
    end
end

function Library:LoadConfig(configName)
    if not readfile or not isfile then return end
    local filePath = GetConfigFolder(configName) .. ".json"
    if not isfile(filePath) then
        -- fallback to AcrylicConfigs
        filePath = "AcrylicConfigs/" .. configName .. ".json"
        if not isfile(filePath) then
            self:Notify({
                Title = "Config Error",
                Description = "Config file '" .. configName .. "' not found",
                Type = "Error",
                Duration = 2.5
            })
            return
        end
    end
    local ok, content = pcall(readfile, filePath)
    if not ok or not content or content == "" then return end
    local dataOk, data = pcall(function() return hs:JSONDecode(content) end)
    if not dataOk or type(data) ~= "table" then return end

    for id, val in pairs(data) do
        local elem = self._configElements[id]
        if elem then
            if elem.type == "ColorPicker" and type(val) == "table" and #val >= 3 then
                elem.set(Color3.new(val[1], val[2], val[3]))
            elseif elem.type == "Keybind" and type(val) == "string" then
                local keyCode = Enum.KeyCode[val]
                if keyCode then
                    elem.set(keyCode)
                end
            else
                elem.set(val)
            end
        end
    end

    self:Notify({
        Title = "Config Loaded",
        Description = "Loaded configuration '" .. configName .. "'",
        Type = "Success",
        Duration = 2.5
    })
end

function Library:DeleteConfig(configName)
    if not delfile or not isfile then return end
    local path = GetConfigFolder(configName) .. ".json"
    if isfile(path) then
        pcall(delfile, path)
        self:Notify({
            Title = "Config Deleted",
            Description = "Deleted '" .. configName .. "'",
            Type = "Warning",
            Duration = 2.5
        })
    end
end

function Library:GetConfigs()
    return GetAvailableConfigs()
end

function Library:SetAutoSave(enabled)
    self._autoSave = enabled
end

function Library._SelectTab(lib, tab, btn, stroke, icon, textLabel, activeIndicator)
    if lib.currentTab then
        lib.currentTab.content.Visible = false
        CreateTween(lib.currentTab.button, {BackgroundTransparency = 1}, animationspeed.Fast)
        CreateTween(lib.currentTab.icon, {ImageColor3 = c.TextDark}, animationspeed.Fast)
        CreateTween(lib.currentTab.textLabel, {TextColor3 = c.TextDark}, animationspeed.Fast)
        lib.currentTab.stroke.Transparency = 1
        if lib.currentTab.activeIndicator then
            lib.currentTab.activeIndicator.Visible = false
        end
    end

    lib.currentTab = tab
    tab.content.Visible = true

    CreateTween(btn, {BackgroundTransparency = 0}, animationspeed.Fast)
    CreateTween(icon, {ImageColor3 = c.Accent}, animationspeed.Fast)
    CreateTween(textLabel, {TextColor3 = c.Text}, animationspeed.Fast)
    stroke.Color = c.BorderLit
    stroke.Transparency = 0
    if activeIndicator then
        activeIndicator.Visible = true
    end
end

function Library._CreateToggle(tab, config)
    local name = config.Name or "Toggle"
    local default = config.Default or false
    local callback = config.Callback or function() end
    local flag = config.Flag
    local tooltip = config.Tooltip
    local keybindCode = config.Keybind
    local colorPickerVal = config.Color
    local enabled = default

    local frame = CreateInstance("Frame", {
        Name = "Toggle_" .. name,
        BackgroundColor3 = c.Secondary,
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 0, s.Button.Height),
        Parent = tab.content
    })
    CreateCorner(frame, 6)
    local stroke = CreateStroke(frame, c.Border, 0, 1)

    local nameLabel = CreateInstance("TextLabel", {
        Name = "Name",
        FontFace = f.Regular,
        TextColor3 = c.Text,
        Text = name,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0.5, -9),
        TextSize = textsize.Normal,
        Size = UDim2.new(1, -120, 0, 18),
        Parent = frame
    })

    local switchBg = CreateInstance("Frame", {
        Name = "SwitchBackground",
        BackgroundColor3 = enabled and c.Toggle.Enabled or c.Toggle.Disabled,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -10, 0.5, 0),
        BorderSizePixel = 0,
        Size = UDim2.new(0, s.Toggle.Width, 0, s.Toggle.Height),
        Parent = frame
    })
    CreateCorner(switchBg, 100)

    local switchCircle = CreateInstance("Frame", {
        Name = "Circle",
        BackgroundColor3 = enabled and c.Toggle.Circle or c.Toggle.CircleOff,
        AnchorPoint = Vector2.new(0, 0.5),
        Position = enabled and UDim2.new(0, s.Toggle.Width - s.Toggle.Circle - 3, 0.5, 0) or UDim2.new(0, 3, 0.5, 0),
        BorderSizePixel = 0,
        Size = UDim2.new(0, s.Toggle.Circle, 0, s.Toggle.Circle),
        Parent = switchBg
    })
    CreateCorner(switchCircle, 100)

    local button = CreateInstance("TextButton", {
        Name = "Button",
        Text = "",
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        Parent = frame
    })

    local function UpdateToggle()
        if enabled then
            CreateTween(switchBg, {BackgroundColor3 = c.Toggle.Enabled}, animationspeed.Normal)
            CreateTween(switchCircle, {
                Position = UDim2.new(0, s.Toggle.Width - s.Toggle.Circle - 3, 0.5, 0),
                BackgroundColor3 = c.Toggle.Circle
            }, animationspeed.Normal)
        else
            CreateTween(switchBg, {BackgroundColor3 = c.Toggle.Disabled}, animationspeed.Normal)
            CreateTween(switchCircle, {
                Position = UDim2.new(0, 3, 0.5, 0),
                BackgroundColor3 = c.Toggle.CircleOff
            }, animationspeed.Normal)
        end
    end

    button.MouseEnter:Connect(function()
        CreateTween(frame, {BackgroundColor3 = c.CardHover}, animationspeed.Fast)
        CreateTween(stroke, {Color = c.BorderLit}, animationspeed.Fast)
    end)

    button.MouseLeave:Connect(function()
        CreateTween(frame, {BackgroundColor3 = c.Secondary}, animationspeed.Fast)
        CreateTween(stroke, {Color = c.Border}, animationspeed.Fast)
    end)

    button.MouseButton1Click:Connect(function()
        enabled = not enabled
        UpdateToggle()
        callback(enabled)
    end)

    -- Embedded keybind support
    if keybindCode then
        local boundKey = keybindCode
        local keyBox = CreateInstance("Frame", {
            Name = "KeyBox",
            BackgroundColor3 = c.Background,
            AnchorPoint = Vector2.new(1, 0.5),
            Position = UDim2.new(1, -(s.Toggle.Width + 18), 0.5, 0),
            Size = UDim2.new(0, 24, 0, 20),
            AutomaticSize = Enum.AutomaticSize.X,
            Parent = frame
        })
        CreateCorner(keyBox, 4)
        CreateStroke(keyBox, c.Border, 0, 1)
        CreatePadding(keyBox, 2, 2, 6, 6)

        local keyText = CreateInstance("TextLabel", {
            Name = "Text",
            FontFace = f.SemiBold,
            TextColor3 = c.TextDark,
            Text = boundKey.Name,
            TextSize = textsize.Tiny,
            BackgroundTransparency = 1,
            AutomaticSize = Enum.AutomaticSize.X,
            Size = UDim2.new(0, 0, 1, 0),
            Parent = keyBox
        })

        local keyBtn = CreateInstance("TextButton", {
            Text = "",
            BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 1, 0),
            Parent = keyBox
        })

        local listening = false
        keyBtn.MouseButton1Click:Connect(function()
            listening = true
            keyText.Text = "..."
        end)

        local conn = ui.InputBegan:Connect(function(input, gp)
            if listening and input.UserInputType == Enum.UserInputType.Keyboard then
                boundKey = input.KeyCode
                listening = false
                keyText.Text = boundKey.Name
            elseif (not gp) and input.KeyCode == boundKey then
                enabled = not enabled
                UpdateToggle()
                callback(enabled)
            end
        end)
        table.insert(Connections, conn)
    end

    if tooltip then
        AttachTooltip(button, tooltip, tab._library.screenGui)
    end

    local methods = {
        SetValue = function(_, value)
            enabled = value
            UpdateToggle()
            callback(enabled)
        end,
        GetValue = function()
            return enabled
        end
    }

    if flag and tab._library then
        tab._library:_RegisterConfigElement(flag, "Toggle",
            function() return enabled end,
            function(value) methods:SetValue(value) end
        )
    end

    table.insert(tab.elements, {name = name, frame = frame})

    return methods
end

    local name = config.Name or "Slider"
    local min = config.Min or 0
    local max = config.Max or 100
    local default = config.Default or min
    local step = config.Step or 1
    local decimals = config.Decimals or 0

    local callback = config.Callback or function() end

    local currentValue = default

    local frame = CreateInstance("Frame", {

        Parent = frame
    })

    local sliderBg = CreateInstance("Frame", {
        Name = "SliderBackground",

        Size = UDim2.new(1, -24, 0, 6),
        Parent = frame
    })
    CreateCorner(sliderBg, 100)

    local initialProgress = math.clamp((default - min) / (max - min), 0, 1)
    local sliderFill = CreateInstance("Frame", {
        Name = "SliderFill",
        BackgroundColor3 = c.Accent,
        BorderSizePixel = 0,

        Position = UDim2.new(1, 0, 0.5, 0),
        Size = UDim2.new(0, 12, 0, 12),
        Parent = sliderFill
    })
    CreateCorner(sliderKnob, 100)

    local function RoundValue(val)
        if decimals > 0 then
            local mult = 10 ^ decimals
            return math.floor(val * mult + 0.5) / mult
        else
            return math.floor(val + 0.5)
        end
    end

    local function UpdateSlider(progress, triggerCallback)
        progress = math.clamp(progress, 0, 1)
        local rawVal = min + (max - min) * progress
        if step and step > 0 then
            rawVal = math.floor((rawVal - min) / step + 0.5) * step + min
        end
        currentValue = RoundValue(math.clamp(rawVal, min, max))
        local visualProgress = math.clamp((currentValue - min) / (max - min), 0, 1)
        CreateTween(sliderFill, {Size = UDim2.new(visualProgress, 0, 1, 0)}, animationspeed.SuperFast)
        valueBox.Text = tostring(currentValue) .. suffix
        if triggerCallback ~= false then
            callback(currentValue)
        end
    end

    local dragging = false
    local sliderBtn = CreateInstance("TextButton", {
        Name = "SliderButton",
        Text = "",
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 0),
        Size = UDim2.new(1, 0, 1, 0),
        Parent = sliderBg
    })

    sliderBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            local progress = (input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X
            UpdateSlider(progress, true)
        end
    end)

    local dragEndConn = ui.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    table.insert(Connections, dragEndConn)

    local dragMoveConn = ui.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local progress = (input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X
            UpdateSlider(progress, true)
        end
    end)
    table.insert(Connections, dragMoveConn)

    -- Manual text input
    valueBox.FocusLost:Connect(function()
        local num = tonumber(valueBox.Text:gsub("[^%d%.%-]", ""))
        if num then
            local clamped = math.clamp(num, min, max)
            local p = (clamped - min) / (max - min)
            UpdateSlider(p, true)
        else
            valueBox.Text = tostring(currentValue) .. suffix
        end
    end)

    frame.MouseEnter:Connect(function()
        CreateTween(frame, {BackgroundColor3 = c.CardHover}, animationspeed.Fast)
        CreateTween(stroke, {Color = c.BorderLit}, animationspeed.Fast)
    end)

    frame.MouseLeave:Connect(function()
        CreateTween(frame, {BackgroundColor3 = c.Secondary}, animationspeed.Fast)
        CreateTween(stroke, {Color = c.Border}, animationspeed.Fast)
    end)

    if tooltip then
        AttachTooltip(sliderBtn, tooltip, tab._library.screenGui)
    end

    local methods = {
        SetValue = function(_, value)
            local p = (value - min) / (max - min)
            UpdateSlider(p, true)
        end,
        GetValue = function()
            return currentValue
        end
    }

    if flag and tab._library then
        tab._library:_RegisterConfigElement(flag, "Slider",
            function() return currentValue end,
            function(value) methods:SetValue(value) end
        )
    end

    table.insert(tab.elements, {name = name, frame = frame})

    return methods
end

        Name = "Dropdown_" .. name,
        BackgroundColor3 = c.Secondary,
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 0, s.Dropdown.Height),
        ClipsDescendants = false,
        ZIndex = 1,
        Parent = tab.content
    })

        Name = "Name",
        FontFace = f.Regular,

        Text = name,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0.5, -9),
        TextSize = textsize.Normal,
        Size = UDim2.new(0.45, 0, 0, 18),
        ZIndex = 1,
        Parent = frame
    })

    local selectedDisplay = CreateInstance("Frame", {
        Name = "SelectedDisplay",
        BackgroundColor3 = c.Background,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -10, 0.5, 0),
        BorderSizePixel = 0,
        Size = UDim2.new(0, 140, 0, 24),
        ZIndex = 2,
        Parent = frame
    })

    local displayStroke = CreateStroke(selectedDisplay, c.Border, 0, 1)

        Name = "SelectedLabel",
        FontFace = f.Regular,
        TextColor3 = c.Text,
        Text = multiSelect and (#selected > 0 and table.concat(selected, ", ") or "None") or tostring(selected),
        TextTruncate = Enum.TextTruncate.AtEnd,
        BackgroundTransparency = 1,
        TextSize = textsize.Small,
        Size = UDim2.new(1, -28, 1, 0),
        Position = UDim2.new(0, 8, 0, 0),
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 2,
        Parent = selectedDisplay
    })

        Name = "Arrow",
        Image = LUCIDE.chevron_down,
        ImageColor3 = c.TextDark,
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -18, 0.5, -6),
        Size = UDim2.new(0, 12, 0, 12),
        ZIndex = 2,
        Parent = selectedDisplay
    })

    local maxVisible = 5
    local totalHeight = math.min(#options * s.Dropdown.OptionHeight, maxVisible * s.Dropdown.OptionHeight)

    local optionsContainer = CreateInstance("Frame", {
        Name = "OptionsContainer",
        BackgroundColor3 = c.Secondary,
        BorderSizePixel = 0,
        Size = UDim2.new(0, 140, 0, totalHeight),
        Visible = false,
        ZIndex = 500,
        ClipsDescendants = true,
        Parent = tab._library.screenGui
    })
    CreateCorner(optionsContainer, 6)
    CreateStroke(optionsContainer, c.BorderLit, 0, 1)

        Name = "OptionsScroll",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 1, 0),
        CanvasSize = UDim2.new(0, 0, 0, #options * s.Dropdown.OptionHeight),
        ScrollBarThickness = 2,
        ScrollBarImageColor3 = c.BorderLit,
        ZIndex = 501,
        Parent = optionsContainer
    })
    CreateListLayout(optionsScroll, 0, Enum.SortOrder.LayoutOrder)

    local function UpdateSelectedText()
        if multiSelect then
            selectedLabel.Text = #selected > 0 and table.concat(selected, ", ") or "None"
        else
            selectedLabel.Text = tostring(selected)
        end
    end

    local function PositionOptions()
        if frame and frame.Parent and optionsContainer then
            local absPos = selectedDisplay.AbsolutePosition
            local absSize = selectedDisplay.AbsoluteSize
            optionsContainer.Position = UDim2.new(0, absPos.X, 0, absPos.Y + absSize.Y + 4)
        end
    end

    local function CreateOptionButton(option)
        local isSelected = multiSelect and table.find(selected, option) ~= nil or (selected == option)
        local optBtn = CreateInstance("TextButton", {
            Name = option,
            FontFace = f.Regular,
            TextColor3 = isSelected and c.Accent or c.Text,
            Text = option,
            BackgroundColor3 = c.Background,
            BackgroundTransparency = isSelected and 0.2 or 1,
            BorderSizePixel = 0,
            TextSize = textsize.Small,
            Size = UDim2.new(1, 0, 0, s.Dropdown.OptionHeight),
            ZIndex = 502,
            Parent = optionsScroll
        })

        optBtn.MouseEnter:Connect(function()
            CreateTween(optBtn, {BackgroundTransparency = 0.4}, animationspeed.Fast)
        end)

        optBtn.MouseLeave:Connect(function()
            local curSelected = multiSelect and table.find(selected, option) ~= nil or (selected == option)
            CreateTween(optBtn, {BackgroundTransparency = curSelected and 0.2 or 1}, animationspeed.Fast)
        end)

        optBtn.MouseButton1Click:Connect(function()
            if multiSelect then
                local idx = table.find(selected, option)
                if idx then
                    table.remove(selected, idx)
                    optBtn.TextColor3 = c.Text
                    optBtn.BackgroundTransparency = 1
                else
                    table.insert(selected, option)
                    optBtn.TextColor3 = c.Accent
                    optBtn.BackgroundTransparency = 0.2
                end
                UpdateSelectedText()
                callback(selected)
            else
                selected = option
                UpdateSelectedText()
                callback(selected)
                expanded = false
                optionsContainer.Visible = false
                CreateTween(arrow, {Rotation = 0}, animationspeed.Normal)
            end
        end)

        return optBtn
    end

    for _, opt in ipairs(options) do
        CreateOptionButton(opt)
    end

    local toggleBtn = CreateInstance("TextButton", {
        Name = "ToggleBtn",
        Text = "",
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        ZIndex = 3,
        Parent = selectedDisplay
    })

    toggleBtn.MouseButton1Click:Connect(function()
        for _, other in ipairs(Library._activeDropdowns) do
            if other ~= optionsContainer and other.Visible then
                other.Visible = false
            end
        end

        expanded = not expanded
        if expanded then
            PositionOptions()
            optionsContainer.Visible = true
            CreateTween(arrow, {Rotation = 180}, animationspeed.Normal)
        else
            optionsContainer.Visible = false
            CreateTween(arrow, {Rotation = 0}, animationspeed.Normal)
        end
    end)

    table.insert(Library._activeDropdowns, optionsContainer)

    local clickConn = ui.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 and expanded then
            task.wait(0.05)
            if not expanded then return end
            local mPos = ui:GetMouseLocation()
            local cPos = optionsContainer.AbsolutePosition
            local cSize = optionsContainer.AbsoluteSize
            local dPos = selectedDisplay.AbsolutePosition
            local dSize = selectedDisplay.AbsoluteSize

            local inContainer = mPos.X >= cPos.X and mPos.X <= cPos.X + cSize.X and mPos.Y >= cPos.Y and mPos.Y <= cPos.Y + cSize.Y
            local inDisplay = mPos.X >= dPos.X and mPos.X <= dPos.X + dSize.X and mPos.Y >= dPos.Y and mPos.Y <= dPos.Y + dSize.Y

            if not inContainer and not inDisplay then
                expanded = false
                optionsContainer.Visible = false
                CreateTween(arrow, {Rotation = 0}, animationspeed.Normal)
            end
        end
    end)
    table.insert(Connections, clickConn)

    frame.Destroying:Connect(function()
        if optionsContainer then
            optionsContainer:Destroy()
        end
    end)

    if tooltip then
        AttachTooltip(toggleBtn, tooltip, tab._library.screenGui)
    end

    local methods = {
        SetValue = function(_, value)
            selected = value
            UpdateSelectedText()
            callback(selected)
        end,
        GetValue = function()
            return selected
        end,
        Refresh = function(_, newOptions)
            options = newOptions
            for _, child in ipairs(optionsScroll:GetChildren()) do
                if child:IsA("TextButton") then
                    child:Destroy()
                end
            end
            for _, opt in ipairs(options) do
                CreateOptionButton(opt)
            end
            optionsScroll.CanvasSize = UDim2.new(0, 0, 0, #options * s.Dropdown.OptionHeight)
            local newH = math.min(#options * s.Dropdown.OptionHeight, maxVisible * s.Dropdown.OptionHeight)
            optionsContainer.Size = UDim2.new(0, 140, 0, newH)
        end
    }

    if flag and tab._library then
        tab._library:_RegisterConfigElement(flag, "Dropdown",
            function() return selected end,
            function(value) methods:SetValue(value) end
        )
    end

    table.insert(tab.elements, {name = name, frame = frame})

    return methods
end

            ColorSequenceKeypoint.new(0.833, Color3.fromHSV(0.833, 1, 1)),
            ColorSequenceKeypoint.new(1, Color3.fromHSV(1, 1, 1))
        }),
        Parent = hueSlider
    })

    local hueCursor = CreateInstance("Frame", {
        Name = "HueCursor",
        BackgroundColor3 = Color3.new(1, 1, 1),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(hue, 0, 0.5, 0),
        Size = UDim2.new(0, 10, 0, 10),
        ZIndex = 3005,
        Parent = hueSlider
    })
    CreateCorner(hueCursor, 100)
    CreateStroke(hueCursor, Color3.new(0, 0, 0), 0, 1)

    local function UpdateColor()
        currentColor = Color3.fromHSV(hue, sat, val)
        colorPreview.BackgroundColor3 = currentColor
        svPicker.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
        svCursor.Position = UDim2.new(sat, 0, 1 - val, 0)
        hueCursor.Position = UDim2.new(hue, 0, 0.5, 0)
        callback(currentColor)
    end

    local svDragging, hueDragging = false, false
    local function ProcessInput(input)
        if not pickerContainer.Visible then return end
        if svDragging then
            local size = svPicker.AbsoluteSize
            local pos = svPicker.AbsolutePosition
            sat = math.clamp((input.Position.X - pos.X) / size.X, 0, 1)
            val = 1 - math.clamp((input.Position.Y - pos.Y) / size.Y, 0, 1)
            UpdateColor()
        elseif hueDragging then
            local size = hueSlider.AbsoluteSize
            local pos = hueSlider.AbsolutePosition
            hue = math.clamp((input.Position.X - pos.X) / size.X, 0, 1)
            UpdateColor()
        end
    end

    svPicker.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            svDragging = true
            ProcessInput(input)
        end
    end)

    hueSlider.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            hueDragging = true
            ProcessInput(input)
        end
    end)

    table.insert(Connections, ui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            ProcessInput(input)
        end
    end))

    table.insert(Connections, ui.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            svDragging = false
            hueDragging = false
        end
    end))

    local function ClosePicker()
        pickerContainer.Visible = false
        expanded = false
        if Library.ActivePicker == ClosePicker then
            Library.ActivePicker = nil
        end
    end

    local function OpenPicker()
        if Library.ActivePicker then
            Library.ActivePicker()
        end
        Library.ActivePicker = ClosePicker

        local btnPos = colorPreview.AbsolutePosition
        local viewport = workspace.CurrentCamera.ViewportSize
        local targetX = btnPos.X - 180
        local targetY = btnPos.Y

        if targetY + 130 > viewport.Y then
            targetY = viewport.Y - 140
        end
        if targetX < 0 then
            targetX = btnPos.X + 45
        end

        pickerContainer.Position = UDim2.new(0, targetX, 0, targetY)
        pickerContainer.Visible = true
        expanded = true
    end

    previewBtn.MouseButton1Click:Connect(function()
        if expanded then
            ClosePicker()
        else
            OpenPicker()
        end
    end)

    if tooltip then
        AttachTooltip(previewBtn, tooltip, tab._library.screenGui)
    end

    local methods = {
        SetColor = function(_, color)
            currentColor = color
            hue, sat, val = color:ToHSV()
            UpdateColor()
        end,
        GetColor = function()
            return currentColor
        end
    }

    if flag and tab._library then
        tab._library:_RegisterConfigElement(flag, "ColorPicker",
            function() return currentColor end,
            function(value) methods:SetColor(value) end
        )
    end

    table.insert(tab.elements, {name = name, frame = frame})

    return methods
end

    local stroke = CreateStroke(frame, c.Border, 0, 1)

    local nameLabel = CreateInstance("TextLabel", {
        Name = "Name",
        FontFace = f.Regular,
        TextColor3 = c.Text,
        Text = name,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0.5, -9),
        TextSize = textsize.Normal,
        Size = UDim2.new(0.45, 0, 0, 18),
        Parent = frame
    })

    local textBoxContainer = CreateInstance("Frame", {
        Name = "TextBoxContainer",
        BackgroundColor3 = c.Background,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -10, 0.5, 0),
        BorderSizePixel = 0,
        Size = UDim2.new(0, s.TextBox.InputWidth, 0, 24),
        Parent = frame
    })
    CreateCorner(textBoxContainer, 5)
    local textBoxStroke = CreateStroke(textBoxContainer, c.Border, 0, 1)

    local textBox = CreateInstance("TextBox", {
        Name = "Input",
        FontFace = f.Regular,
        TextColor3 = c.Text,
        PlaceholderText = placeholder,
        PlaceholderColor3 = c.TextFade,
        Text = currentText,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
        BackgroundTransparency = 1,
        TextSize = textsize.Small,
        Size = UDim2.new(1, -12, 1, 0),
        Position = UDim2.new(0, 6, 0, 0),
        ClearTextOnFocus = clearOnFocus,
        Parent = textBoxContainer
    })

    textBox.Focused:Connect(function()
        CreateTween(textBoxStroke, {Color = c.Accent}, animationspeed.Fast)
    end)

    textBox.FocusLost:Connect(function(enterPressed)
        CreateTween(textBoxStroke, {Color = c.Border}, animationspeed.Fast)
        if numbersOnly then
            local numValue = tonumber(textBox.Text)
            if numValue then
                currentText = tostring(numValue)
                textBox.Text = currentText
            else
                textBox.Text = currentText
            end
        else
            currentText = textBox.Text
        end
        callback(currentText, enterPressed)
    end)

    if numbersOnly then
        textBox:GetPropertyChangedSignal("Text"):Connect(function()
            local text = textBox.Text
            local filtered = text:gsub("[^%d%.%-]", "")
            if text ~= filtered then
                textBox.Text = filtered
            end
        end)
    end

    if tooltip then
        AttachTooltip(textBox, tooltip, tab._library.screenGui)
    end

    local methods = {
        SetText = function(_, text)
            currentText = tostring(text)
            textBox.Text = currentText
        end,
        GetText = function()
            return currentText
        end,
        SetPlaceholder = function(_, newPlaceholder)
            textBox.PlaceholderText = newPlaceholder
        end,
        Focus = function()
            textBox:CaptureFocus()
        end
    }

    if flag and tab._library then
        tab._library:_RegisterConfigElement(flag, "TextBox",
            function() return currentText end,
            function(value) methods:SetText(value) end
        )
    end

    table.insert(tab.elements, {name = name, frame = frame})

    return methods
end

function Library._CreateConfigSection(tab)
    local lib = tab._library
    Library._CreateContentSection(tab, "Configuration Profile")

    local configNameBox = Library._CreateTextBox(tab, {
        Name = "Config Profile Name",
        Default = "default",
        Placeholder = "Enter profile name...",
        Callback = function(text)
            lib._currentConfig = text
        end
    })

    local configDropdown
    configDropdown = Library._CreateDropdown(tab, {
        Name = "Available Profiles",
        Options = lib:GetConfigs(),
        Default = "default",
        Callback = function(selected)
            configNameBox:SetText(selected)
            lib._currentConfig = selected
        end
    })

    Library._CreateDualButton(tab, {
        Button1 = {
            Name = "Save Config",
            Callback = function()
                local configName = configNameBox:GetText()
                if configName and configName ~= "" then
                    lib:SaveConfig(configName)
                    configDropdown:Refresh(lib:GetConfigs())
                end
            end
        },
        Button2 = {
            Name = "Load Config",
            Callback = function()
                local configName = configNameBox:GetText()
                if configName and configName ~= "" then
                    lib:LoadConfig(configName)
                end
            end
        }
    })

    Library._CreateDualButton(tab, {
        Button1 = {
            Name = "Delete Profile",
            Callback = function()
                local configName = configNameBox:GetText()
                if configName and configName ~= "" then
                    lib:Prompt({
                        Title = "Delete Profile",
                        Description = "Are you sure you want to delete profile '" .. configName .. "'?",
                        ConfirmText = "Delete",
                        CancelText = "Cancel",
                        OnConfirm = function()
                            lib:DeleteConfig(configName)
                            configDropdown:Refresh(lib:GetConfigs())
                        end
                    })
                end
            end
        },
        Button2 = {
            Name = "Refresh List",
            Callback = function()
                configDropdown:Refresh(lib:GetConfigs())
                lib:Notify({
                    Title = "Profiles Refreshed",
                    Description = "Profile list successfully updated",
                    Type = "Info",
                    Duration = 2
                })
            end
        }
    })

    Library._CreateToggle(tab, {
        Name = "Auto-Save on Changes",
        Default = false,
        Callback = function(enabled)
            lib:SetAutoSave(enabled)
        end
    })

    return {
        RefreshConfigs = function()
            configDropdown:Refresh(lib:GetConfigs())
        end
    }
end

return Library
