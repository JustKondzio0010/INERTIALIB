-- ─── Helpers ─────────────────────────────────────────────────────────────────
local function Tween(obj, props, t, style, dir)
    TweenService:Create(obj, TweenInfo.new(t or 0.25, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out), props):Play()
end

local function MakeDraggable(handle, frame)
    local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = inp.Position
            startPos  = frame.Position
            inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
            dragInput = inp
        end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if inp == dragInput and dragging then
            local delta = inp.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function MakeCorner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = parent
    return c
end

local function MakeStroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Divider
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = parent
    return s
end

local function MakePadding(parent, top, bottom, left, right)
    local p = Instance.new("UIPadding")
    p.PaddingTop    = UDim.new(0, top    or 0)
    p.PaddingBottom = UDim.new(0, bottom or 0)
    p.PaddingLeft   = UDim.new(0, left   or 0)
    p.PaddingRight  = UDim.new(0, right  or 0)
    p.Parent = parent
    return p
end

            if typeof(decoded.uiScale)       == "number"  then prefs.uiScale       = decoded.uiScale       end
            if typeof(decoded.reduceEffects) == "boolean" then prefs.reduceEffects = decoded.reduceEffects end

    local flags, flagBindings = {}, {}
    local function registerFlag(flag, setter)
        if type(flag)=="string" and flag~="" and type(setter)=="function" then flagBindings[flag]=setter end
    end
    local function setFlagValue(flag, value, apply)
        if type(flag)=="string" and flag~="" then
            flags[flag]=value
            if apply and flagBindings[flag] then pcall(flagBindings[flag], value) end
        end
    end
    local function snapshotFlags()
        local out={}; for k,v in next,flags do out[k]=v end; return out
    end
    local function sanitizeConfigName(name)
        name=tostring(name or ""):gsub("[^%w%-%_ ]",""):gsub("%s+"," "):gsub("^%s+",""):gsub("%s+$","")
        if name=="" then name="default" end; return name
    end
    local function makeFlagKey(prefix, name)
        local k=(tostring(prefix or "tab")..".."..tostring(name or "flag")):gsub("%s+","_"):gsub("[^%w%._%-]","")
        if k=="" then k="flag" end; return k
    end

    -- configs

    local function ensureConfigFolder()
        if typeof(isfolder)=="function" and typeof(makefolder)=="function" and not isfolder(configFolder) then
            pcall(function() makefolder(configFolder) end)
        end
    end
    local function listConfigs()
        ensureConfigFolder()
        if typeof(listfiles)~="function" then return {} end
        local files={}
        local ok,res=pcall(function() return listfiles(configFolder) end)
        if not ok or type(res)~="table" then return {} end
        for _,p in ipairs(res) do
            local n=tostring(p):match("([^\\/]+)%.json$")
            if n and n~="" then table.insert(files, n) end
        end
        table.sort(files); return files
    end
    local function configPath(name) return configFolder.."/"..sanitizeConfigName(name)..".json" end
    local function saveConfig(name)
        ensureConfigFolder()
        if typeof(writefile)~="function" then return false,"no_writefile" end
        local ok,body=pcall(function() return HttpService:JSONEncode({flags=snapshotFlags()}) end)
        if not ok then return false,"encode_failed" end
        local ok2=pcall(function() writefile(configPath(name),body) end)
        if not ok2 then return false,"write_failed" end; return true
    end
    local function loadConfig(name)
        ensureConfigFolder()
        if typeof(readfile)~="function" or typeof(isfile)~="function" then return false,"no_readfile" end
        local path=configPath(name)
        if not isfile(path) then return false,"not_found" end
        local ok,decoded=pcall(function() return HttpService:JSONDecode(readfile(path)) end)
        if not ok or type(decoded)~="table" or type(decoded.flags)~="table" then return false,"decode_failed" end
        for k,v in next,decoded.flags do setFlagValue(k,v,true) end; return true
    end
    local function deleteConfig(name)
        ensureConfigFolder()
        if typeof(delfile)~="function" then return false,"no_delfile" end
        local ok=pcall(function() delfile(configPath(name)) end)
        if not ok then return false,"delete_failed" end; return true
    end

    local keyExpiresAt = nil
    local function formatKeyTimeLeft(s)
        s = math.max(0, math.floor(tonumber(s) or 0))
        local h = math.floor(s/3600)
        local m = math.floor((s%3600)/60)
        if h<=0 and m<=0 then return "0m" end
        if h<=0 then return m.."m" end
        return h.."h "..m.."m"
    end
    local function updateKeyBadge()
        if not keyExpiresAt then
            KeyBadgeText.Text = "NO KEY"
            KeyBadgeText.TextColor3 = Theme.TextMuted
            return
        end
        local left = keyExpiresAt - os.time()
        if left <= 0 then
            KeyBadgeText.Text = "⚠ EXPIRED"
            KeyBadgeText.TextColor3 = Theme.Danger
            return
        end
        KeyBadgeText.Text = "🔑 " .. formatKeyTimeLeft(left)
        KeyBadgeText.TextColor3 = Theme.TextPrimary
    end

            -- track
            local Track = Instance.new("Frame")
            Track.Parent = el

            Track.Position = UDim2.new(0, 34, 0, 46)

            Fill.BackgroundColor3 = PresetColor
            Fill.BorderSizePixel = 0
            Fill.Size = UDim2.new((current - min)/(max - min), 0, 1, 0)
            MakeCorner(Fill, 2)

            -- thumb
            local Thumb = Instance.new("Frame")
            Thumb.Parent = Track
            Thumb.AnchorPoint = Vector2.new(0.5, 0.5)
            Thumb.BackgroundColor3 = Theme.White
            Thumb.BorderSizePixel = 0

            Thumb.Size = UDim2.new(0, 14, 0, 14)
            MakeCorner(Thumb, 7)
            MakeStroke(Thumb, PresetColor, 2, 0)

                pct = math.clamp(pct, 0, 1)
                current = math.floor(pct * (max - min) + min)
                Fill.Size = UDim2.new(pct, 0, 1, 0)
                Thumb.Position = UDim2.new(pct, 0, 0.5, 0)
                ValLbl.Text = tostring(current)
                setFlagValue(flag, current, false)
                pcall(callback, current)

            setSliderValue((current - min)/(max - min))

                    setSliderValue(pct)
                end
            end)
            SliderBtn.InputEnded:Connect(function(inp)

                    dragging = false
                end
            end)

                    local pct = math.clamp((inp.Position.X - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
                    setSliderValue(pct)
                end
            end)

            el.MouseEnter:Connect(function() Tween(el, {BackgroundColor3 = Theme.CardHover}, 0.2) end)
            el.MouseLeave:Connect(function() Tween(el, {BackgroundColor3 = Theme.Card}, 0.2) end)

            setFlagValue(flag, current, false)
            registerFlag(flag, function(v) setSliderValue((tonumber(v) or min - min)/(max - min)) end)

            function SliderFunc:Change(val)
                val = math.clamp(tonumber(val) or min, min, max)
                setSliderValue((val - min)/(max - min))
            end

            Container.CanvasSize = UDim2.new(0,0,0, ContainerLayout.AbsoluteContentSize.Y + 10)
            return SliderFunc
        end

                ChkMark.ImageColor3 = PresetColor
                ChkMark.ImageTransparency = 1
                ChkMark.ZIndex = 11

                local function selectItem()
                    selected = itemText
                    SelLbl.Text = itemText
                    SelLbl.TextColor3 = PresetColor
                    -- reset all items
                    for _, child in next, ListFrame:GetChildren() do
                        if child:IsA("TextButton") then
                            if child:FindFirstChild("ItemLbl") then
                                Tween(child.ItemLbl, {TextColor3 = Theme.TextMuted}, 0.2)
                            end
                            if child:FindFirstChild("Frame") then -- ItemDot
                                Tween(child:FindFirstChild("Frame"), {BackgroundColor3 = Theme.TextDim}, 0.2)
                            end
                            if child:FindFirstChild("ImageLabel") then -- ChkMark
                                Tween(child:FindFirstChildOfClass("ImageLabel"), {ImageTransparency = 1}, 0.2)
                            end
                        end
                    end
                    -- highlight this
                    Tween(ItemLbl,  {TextColor3 = PresetColor}, 0.2)
                    Tween(ItemDot,  {BackgroundColor3 = PresetColor}, 0.2)
                    Tween(ChkMark,  {ImageTransparency = 0}, 0.2)
                    setFlagValue(flag, itemText, false)
                    pcall(callback, itemText)
                    task.delay(0.15, closeDrop)
                end

                Item.MouseEnter:Connect(function()
                    Tween(Item, {BackgroundTransparency = 0, BackgroundColor3 = Theme.Card}, 0.15)
                    Tween(ItemLbl, {TextColor3 = Theme.TextPrimary}, 0.15)
                end)
                Item.MouseLeave:Connect(function()
                    Tween(Item, {BackgroundTransparency = 1}, 0.15)
                    Tween(ItemLbl, {TextColor3 = selected == itemText and PresetColor or Theme.TextMuted}, 0.15)
                end)
                Item.MouseButton1Click:Connect(selectItem)

                    ListFrame:TweenSize(UDim2.new(1,0,0, calcOpenH()-headerH),"Out","Quart",0.25,true)
                end
            end

            -- pre-populate
            if list then
                for _, v in ipairs(list) do addItem(tostring(v)) end
            end

            function DropFunc:Add(t)    addItem(tostring(t)) end
            function DropFunc:Clear()
                for _, c in next, ListFrame:GetChildren() do if c:IsA("TextButton") then c:Destroy() end end
                itemCount = 0; selected = nil; SelLbl.Text = "Select..."; SelLbl.TextColor3 = Theme.TextMuted
                if isOpen then closeDrop() end
            end
            function DropFunc:Set(v)
                selected = tostring(v); SelLbl.Text = selected; SelLbl.TextColor3 = PresetColor
                setFlagValue(flag, selected, false); pcall(callback, selected)
            end
            if flag then
                setFlagValue(flag, selected, false)
                registerFlag(flag, function(v) DropFunc:Set(v) end)
            end

            Container.CanvasSize = UDim2.new(0,0,0, ContainerLayout.AbsoluteContentSize.Y + 10)
            return DropFunc
        end

            KeyBtn.AutoButtonColor = false
            MakeCorner(KeyBtn, 6)
            MakeStroke(KeyBtn, Theme.Divider, 1, 0)

            KeyBtn.MouseButton1Click:Connect(function()
                if listening then return end
                listening = true
                KeyBtn.Text = "[ ... ]"
                Tween(KeyBtn, {TextColor3 = Theme.TextMuted}, 0.1)
                local conn
                conn = UserInputService.InputBegan:Connect(function(inp, gp)
                    if gp then return end
                    if inp.UserInputType == Enum.UserInputType.Keyboard then
                        currentKey = inp.KeyCode.Name
                        KeyBtn.Text = "[" .. currentKey .. "]"
                        Tween(KeyBtn, {TextColor3 = PresetColor}, 0.2)
                        listening = false
                        conn:Disconnect()
                    end
                end)
            end)

            UserInputService.InputBegan:Connect(function(inp, gp)
                if not gp and inp.UserInputType == Enum.UserInputType.Keyboard and inp.KeyCode.Name == currentKey then
                    pcall(callback)
                end
            end)

            el.MouseEnter:Connect(function() Tween(el, {BackgroundColor3 = Theme.CardHover}, 0.2) end)
            el.MouseLeave:Connect(function() Tween(el, {BackgroundColor3 = Theme.Card}, 0.2) end)

            Container.CanvasSize = UDim2.new(0,0,0, ContainerLayout.AbsoluteContentSize.Y + 10)
        end

            HueGrad.Color = ColorSequence.new{
                ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255,0,4)),
                ColorSequenceKeypoint.new(0.20, Color3.fromRGB(234,255,0)),
