local function draggable(handle, frame)
    local drag, di, ds, sp = false, nil, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            drag = true; ds = i.Position; sp = frame.Position
            i.Changed:Connect(function() if i.UserInputState == Enum.UserInputState.End then drag = false end end)
        end
    end)
    handle.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then di = i end
    end)
    UIS.InputChanged:Connect(function(i)
        if i == di and drag then
            local d = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
        end
    end)
end

local function hexToColor(hex)
    if not hex then return nil end
    hex = hex:gsub("#", "")
    if #hex == 6 then
        local r, g, b = tonumber(hex:sub(1,2),16), tonumber(hex:sub(3,4),16), tonumber(hex:sub(5,6),16)
        if r and g and b then return Color3.fromRGB(r, g, b) end
    end
    return nil
end

local function parseColor(colorVal)
    if typeof(colorVal) == "Color3" then return colorVal, nil end
    if type(colorVal) == "string" then
        if colorVal:find("|") then
            local parts = colorVal:split("|")
            local cols = {}
            for _, p in ipairs(parts) do
                local c = hexToColor(p:gsub("%s",""))
                if c then table.insert(cols, c) end
            end
            if #cols >= 2 then return cols[1], cols end
        end
        local c = hexToColor(colorVal)
        if c then return c, nil end
    end
    return AccentA, nil
end

local function applyTextGradient(label, colors)
    if not colors or #colors < 2 then return end
    local kps = {}
    for i, c in ipairs(colors) do
        table.insert(kps, ColorSequenceKeypoint.new((i-1)/(#colors-1), c))
    end
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(kps)
    g.Rotation = 90
    g.Parent = label
end

        if type(k)=="string" and k~="" then
            flags[k]=v

    end

    local function snapFlags()
        local out={}; for k,v in next,flags do out[k]=v end; return out
    end
    local function sanitizeName(n)
        n=tostring(n or ""):gsub("[^%w%-%_ ]",""):gsub("%s+"," "):gsub("^%s+",""):gsub("%s+$","")
        return n=="" and "default" or n
    end

    local function savePrefs()
        if typeof(writefile)=="function" then
            pcall(function() writefile(prefsFile, HS:JSONEncode(prefs)) end)
        end
    end
    if typeof(isfile)=="function" and typeof(readfile)=="function" and isfile(prefsFile) then
        local ok, d = pcall(function() return HS:JSONDecode(readfile(prefsFile)) end)
        if ok and type(d)=="table" then
            if type(d.uiScale)=="number" then prefs.uiScale=d.uiScale end
            if type(d.closeKey)=="string" then
                local ok2, kc = pcall(function() return Enum.KeyCode[d.closeKey] end)
                if ok2 and kc then closeKey = kc end
            end
        end
    end

    local function ensureFolder()
        if typeof(isfolder)=="function" and typeof(makefolder)=="function" and not isfolder(configFolder) then
            pcall(function() makefolder(configFolder) end)
        end
    end
    local function cfgPath(n) return configFolder.."/"..sanitizeName(n)..".json" end
    local function saveConfig(n)
        ensureFolder()
        if typeof(writefile)~="function" then return false,"no_writefile" end
        local ok,body = pcall(function() return HS:JSONEncode({flags=snapFlags(),version=1}) end)
        if not ok then return false,"encode_failed" end
        local ok2 = pcall(function() writefile(cfgPath(n),body) end)
        return ok2 or false, ok2 and nil or "write_failed"
    end
    local function loadConfig(n)
        ensureFolder()
        if typeof(readfile)~="function" or typeof(isfile)~="function" then return false,"no_readfile" end
        local path = cfgPath(n)
        if not isfile(path) then return false,"not_found" end
        local ok,d = pcall(function() return HS:JSONDecode(readfile(path)) end)
        if not ok or type(d)~="table" or type(d.flags)~="table" then return false,"decode_failed" end
        for k,v in next,d.flags do setFlag(k,v,true) end
        return true
    end
    local function deleteConfig(n)
        ensureFolder()
        if typeof(delfile)~="function" then return false,"no_delfile" end
        local ok = pcall(function() delfile(cfgPath(n)) end)
        return ok or false, ok and nil or "delete_failed"
    end
    local function listConfigs()
        ensureFolder()
        if typeof(listfiles)~="function" then return {} end
        local ok,res = pcall(function() return listfiles(configFolder) end)
        if not ok or type(res)~="table" then return {} end
        local files={}
        for _,p in ipairs(res) do
            local n=tostring(p):match("([^\\/]+)%.json$")
            if n and n~="" then table.insert(files,n) end
        end
        table.sort(files); return files
    end

                el:TweenSize(UDim2.new(1,0,0, expanded and expandH or 44),"Out","Quart",0.3,true)
                SVSel.Visible = expanded; HueSel.Visible = expanded
                tw(dot, {BackgroundColor3=expanded and AccentA or T.TextLow}, 0.15)
            end)

            SatVal.InputBegan:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 then
                    if cinput then cinput:Disconnect() end
                    cinput = RS.RenderStepped:Connect(function()
                        local x = math.clamp((Mouse.X-SatVal.AbsolutePosition.X)/SatVal.AbsoluteSize.X,0,1)
                        local y = math.clamp((Mouse.Y-SatVal.AbsolutePosition.Y)/SatVal.AbsoluteSize.Y,0,1)
                        SVSel.Position = UDim2.new(x,0,y,0); s=x; v=1-y; updateCP()
                    end)
                end
            end)
            SatVal.InputEnded:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 and cinput then cinput:Disconnect() end
            end)
            HueBar.InputBegan:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 then
                    if hueinput then hueinput:Disconnect() end
                    hueinput = RS.RenderStepped:Connect(function()
                        local y = math.clamp((Mouse.Y-HueBar.AbsolutePosition.Y)/HueBar.AbsoluteSize.Y,0,1)
                        HueSel.Position = UDim2.new(0.5,0,y,0); h=1-y; updateCP()
                    end)
                end
            end)
            HueBar.InputEnded:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 and hueinput then hueinput:Disconnect() end
            end)

            updateCP()
            Container.CanvasSize = UDim2.new(0,0,0, CLayout.AbsoluteContentSize.Y+12)
        end

        function C:Label(opts)
            local ltext = type(opts)=="string" and opts or (opts.text or opts[1] or "")
            local lcolor = type(opts)=="table" and opts.color or nil

            local el = Instance.new("TextLabel")
            el.BackgroundTransparency = 1
            el.Size = UDim2.new(1,0,0,22)
            el.Font = Enum.Font.Gotham
            el.Text = ltext
            el.TextColor3 = T.TextLow
            el.TextSize = 11
            el.TextXAlignment = Enum.TextXAlignment.Left
            el.ZIndex = 3
            el.Parent = Container
            pad(el, 0, 0, 14, 0)
            if lcolor then
                local col, cols = parseColor(lcolor)
                if cols then applyTextGradient(el, cols)
                elseif col then el.TextColor3 = col end
            end
            Container.CanvasSize = UDim2.new(0,0,0, CLayout.AbsoluteContentSize.Y+12)
        end

        function C:Separator(opts)
            local stitle = type(opts)=="string" and opts or (type(opts)=="table" and opts.title or nil)
            local el = Instance.new("Frame")
            el.BackgroundTransparency = 1
            el.Size = UDim2.new(1,0,0, stitle and 22 or 10)
            el.ZIndex = 3
            el.Parent = Container
            local line = Instance.new("Frame")
            line.BackgroundColor3 = T.Border
            line.BorderSizePixel = 0
            line.AnchorPoint = Vector2.new(0.5,0.5)
            line.Position = UDim2.new(0.5,0,0.5,0)
            line.Size = UDim2.new(1,0,0,1)
            line.ZIndex = 3
            line.Parent = el
            if stitle then
                local tlbl2 = Instance.new("TextLabel")
                tlbl2.BackgroundColor3 = T.Card
                tlbl2.AnchorPoint = Vector2.new(0.5,0.5)
                tlbl2.Position = UDim2.new(0.5,0,0.5,0)
                tlbl2.Size = UDim2.new(0,0,0,14)
                tlbl2.AutomaticSize = Enum.AutomaticSize.X
                tlbl2.Font = Enum.Font.Gotham
                tlbl2.Text = "  "..stitle.."  "
                tlbl2.TextColor3 = T.TextLow
                tlbl2.TextSize = 10
                tlbl2.ZIndex = 4
                tlbl2.Parent = el
                pad(tlbl2, 0, 0, 4, 4)
            end
            Container.CanvasSize = UDim2.new(0,0,0, CLayout.AbsoluteContentSize.Y+12)
        end

        return C
    end

    local Settings = Tabs:Tab("Settings", "settings")

    Settings:Slider({
        name="UI Scale", min=65, max=110, default=prefs.uiScale, suffix="%",
        callback=function(v) prefs.uiScale=v; savePrefs(); applyScale() end
    })

    Settings:Bind({
        name="Close Keybind", key=closeKey.Name,
        callback=function() end
    })

    Settings:Toggle({
        name="Mobile Button", default=isTouchDevice,
        callback=function(v) MobileBtn.Visible=v end
    })

    Settings:Separator("Configs")

    local selConfig = nil
    local cfgName   = "default"

    local cfgDrop = Settings:Dropdown({
        name="Select Config", list=listConfigs(),
        callback=function(v) selConfig=v end
    })

    Settings:Textbox({
        name="Config Name", placeholder="default", disappear=false,
        callback=function(v) cfgName=sanitizeName(v) end
    })

    Settings:Button({
        name="Save Config", icon="save",
        callback=function()
            local ok,err = saveConfig(cfgName)
            if not ok then InertiaUI:Notification({title="Error",message="Save failed: "..tostring(err),type="error"}); return end
            selConfig=cfgName; cfgDrop:Clear(); for _,n in ipairs(listConfigs()) do cfgDrop:Add(n) end
            InertiaUI:Notification({title="Saved",message="Config '"..cfgName.."' saved.",type="success"})
        end
    })

    Settings:Button({
        name="Load Config", icon="refresh",
        callback=function()
            local n = selConfig or cfgName
            local ok,err = loadConfig(n)
            if not ok then InertiaUI:Notification({title="Error",message="Load failed: "..tostring(err),type="error"}); return end
            InertiaUI:Notification({title="Loaded",message="Config '"..n.."' loaded.",type="success"})
        end
    })

    Settings:Button({
        name="Delete Config", icon="trash",
        callback=function()
            local n = selConfig or cfgName
            local ok,err = deleteConfig(n)
            if not ok then InertiaUI:Notification({title="Error",message="Delete failed: "..tostring(err),type="error"}); return end
            if selConfig==n then selConfig=nil end
            cfgDrop:Clear(); for _,n2 in ipairs(listConfigs()) do cfgDrop:Add(n2) end
            InertiaUI:Notification({title="Deleted",message="Config '"..n.."' deleted.",type="warn"})
        end
    })

    Settings:Button({
        name="Center UI", icon="home",
        callback=function() Main.Position=UDim2.new(0.5,0,0.5,0) end
    })

    Settings:Button({
        name="Toggle UI", icon="eye",
        callback=function() setVisible(uiHidden) end
    })

    return Tabs
end
